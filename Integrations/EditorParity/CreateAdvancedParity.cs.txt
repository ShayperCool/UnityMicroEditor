if (UnityEditor.EditorApplication.isPlaying) throw new System.InvalidOperationException("AdvancedParity authoring requires Edit Mode.");
if (UnityEditor.AssetDatabase.LoadAssetAtPath<UnityEditor.SceneAsset>("Assets/Scenes/AdvancedParity.unity") != null) throw new System.InvalidOperationException("AdvancedParity already exists; preserve authored scene.");
var scene = UnityEditor.SceneManagement.EditorSceneManager.NewScene(UnityEditor.SceneManagement.NewSceneSetup.EmptyScene, UnityEditor.SceneManagement.NewSceneMode.Single);
var cameraObject = new UnityEngine.GameObject("AdvancedCamera");
var camera = cameraObject.AddComponent<UnityEngine.Camera>();
camera.backgroundColor = new UnityEngine.Color(0.12f, 0.18f, 0.24f, 1);
cameraObject.tag = "MainCamera";
cameraObject.transform.position = new UnityEngine.Vector3(0, 0, -10);
var canvasObject = new UnityEngine.GameObject("AdvancedCanvas", typeof(UnityEngine.RectTransform));
var canvas = canvasObject.AddComponent<UnityEngine.Canvas>();
canvas.renderMode = UnityEngine.RenderMode.ScreenSpaceOverlay;
canvasObject.AddComponent<UnityEngine.UI.GraphicRaycaster>();
var probeObject = new UnityEngine.GameObject("AdvancedProbe", typeof(UnityEngine.RectTransform));
probeObject.transform.SetParent(canvasObject.transform, false);
var rect = probeObject.GetComponent<UnityEngine.RectTransform>();
rect.sizeDelta = new UnityEngine.Vector2(250, 80);
probeObject.AddComponent<UnityEngine.UI.Image>().color = new UnityEngine.Color(0.2f, 0.7f, 0.35f, 1);
probeObject.AddComponent<UnityEngine.UI.Button>();
probeObject.AddComponent<AgentParity.AdvancedAgentProbe>();
probeObject.AddComponent<UnityEngine.AudioSource>().playOnAwake = false;
probeObject.AddComponent<UnityEngine.Animator>();
var eventsObject = new UnityEngine.GameObject("AdvancedEventSystem");
eventsObject.AddComponent<UnityEngine.EventSystems.EventSystem>();
eventsObject.AddComponent<UnityEngine.InputSystem.UI.InputSystemUIInputModule>();
UnityEditor.SceneManagement.EditorSceneManager.MarkSceneDirty(scene);
if (!UnityEditor.SceneManagement.EditorSceneManager.SaveScene(scene, "Assets/Scenes/AdvancedParity.unity")) throw new System.InvalidOperationException("AdvancedParity save failed.");
UnityEditor.AssetDatabase.SaveAssets();
return new { scenePath = scene.path, rootCount = scene.rootCount, probePath = "AdvancedCanvas/AdvancedProbe", unityVersion = UnityEngine.Application.unityVersion };
